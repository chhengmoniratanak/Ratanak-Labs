! =====================================================================
! R1-PE CONFIGURATION
! =====================================================================
enable
configure terminal
hostname R1-PE
no ip domain-lookup

vrf definition CUST_1
 rd 1:1
 route-target export 1:1
 route-target import 1:1
 address-family ipv4
 exit-address-family

vrf definition CUST_2
 rd 2:2
 route-target export 2:2
 route-target import 2:2
 address-family ipv4
 exit-address-family

mpls ip
mpls label protocol ldp

interface Loopback0
 ip address 1.1.1.1 255.255.255.255

interface GigabitEthernet0/0
 ip address 10.12.0.1 255.255.255.0
 mpls ip
 no shutdown

interface GigabitEthernet1/0
 vrf forwarding CUST_1
 ip address 172.16.1.254 255.255.255.0
 no shutdown

interface GigabitEthernet2/0
 vrf forwarding CUST_2
 ip address 172.16.2.254 255.255.255.0
 no shutdown

router ospf 1
 router-id 1.1.1.1
 network 1.1.1.1 0.0.0.0 area 0
 network 10.12.0.0 0.0.0.255 area 0

router bgp 65000
 bgp router-id 1.1.1.1
 no bgp default ipv4-unicast
 neighbor 4.4.4.4 remote-as 65000
 neighbor 4.4.4.4 update-source Loopback0
 address-family vpnv4
  neighbor 4.4.4.4 activate
  neighbor 4.4.4.4 send-community extended
 exit-address-family
 address-family ipv4 vrf CUST_1
  neighbor 172.16.1.11 remote-as 65001
  neighbor 172.16.1.11 activate
 exit-address-family
 address-family ipv4 vrf CUST_2
  neighbor 172.16.2.22 remote-as 65002
  neighbor 172.16.2.22 activate
 exit-address-family
end
write memory
